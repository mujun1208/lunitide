func TestAuditReviewMCPUninstallDoesNotRevokeChat(t *testing.T){
 ctx:=context.Background();store,err:=storage.OpenTemplated(ctx,filepath.Join(t.TempDir(),"audit.db"));if err!=nil{t.Fatal(err)};defer store.Close()
 repo:=store.AgentRuntimeRepository();e:=NewEngine(nil,"audit");e.m7mcp=m7app.NewMcpRuntimeService(repo);e.mcmarket=mcapp.New(repo)
 e.mcp6Registry=mcp6.NewRegistry(func(context.Context,*mcp6.Endpoint)error{return nil},func(context.Context,*mcp6.Endpoint,string,map[string]any,[]byte)(map[string]any,error){return map[string]any{"ok":true},nil},auditNoSecretLease{})
 e.mcp6Registry.SetDescribeFunc(func(context.Context,*mcp6.Endpoint)(map[string]mcp6.ToolSchema,error){return map[string]mcp6.ToolSchema{"audit_tool":{InputSchema:json.RawMessage(`{"type":"object"}`)}},nil})
 ep,err:=e.m7mcp.Add(ctx,m7app.McpAddInput{Origin:"manual",Transport:"stdio",Command:"npx",Args:[]string{"audit-no-execution"},RiskConfirmed:true});if err!=nil{t.Fatal(err)}
 payload,_:=json.Marshal(map[string]any{"endpointId":ep.EndpointID,"enabled":true})
 if res:=handleMcpToggle(e,ctx,bridge.Request{Payload:payload});!res.OK{t.Fatalf("toggle: %+v",res.Error)}
 token,_,err:=e.mcmarket.IssueConfirmToken(ctx,mcapp.ConfirmMethodUninstall,ep.EndpointID,"");if err!=nil{t.Fatal(err)}
 payload,_=json.Marshal(map[string]any{"endpointId":ep.EndpointID,"confirmToken":token})
 if res:=handleMcConnectorUninstall(e,ctx,bridge.Request{Payload:payload});!res.OK{t.Fatalf("uninstall: %+v",res.Error)}
 listed,err:=e.m7mcp.List(ctx,"");if err!=nil||len(listed)!=1||listed[0].State!="revoked"{t.Fatalf("settings: %+v %v",listed,err)}
 _,err=e.mcp6Registry.Invoke(ctx,chatMcpEndpointID(ep.EndpointID),"audit_tool",nil);if err!=nil{t.Fatalf("bug not reproduced: %v",err)}

}
