package datasourceapp

import (
    "testing"
    "github.com/jackc/pgx/v5"
)

// Parsing only; reserved documentation address; no socket or SQL execution.
func TestAuditPostgresEffectiveHostAuthority(t *testing.T) {
    dsn := "postgres://audit@127.0.0.1/audit?host=192.0.2.10&sslmode=disable"
    cfg, err := pgx.ParseConfig(dsn)
    if err != nil { t.Fatal(err) }
    if cfg.Host != "192.0.2.10" { t.Fatalf("unexpected driver host %q", cfg.Host) }
    if IsLocalDSN("postgres", dsn) {
        t.Fatalf("local-write authorization accepts DSN whose effective driver host is %s", cfg.Host)
    }
}
